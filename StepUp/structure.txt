using StepUp.Application.Exceptions;
using StepUp.Application.Interfaces.Repositories;
using StepUp.Application.Interfaces.Services;
using StepUp.Application.Models;

namespace StepUp.Application.Services;

public class AuthService(IUserRepository userRep,
                         IPasswordHasher passwordHasher,
                         ICurrentUserProvider userProvider,
                         ICurrentUserService userService,
                         IClock clock)
{

    public async Task<User> RegisterAsync(RegisterRequest request, CancellationToken ct = default)
    {
        var passwordErrors = PasswordPolicy.Validate(request.Password);
        if (passwordErrors.Any())
            throw new WeakPasswordException(passwordErrors); 

        if (await userRep.ExistsByEmailAsync(request.Email, ct))
            throw new EmailAlreadyExistsException(request.Email); 

        var user = new User(
            fullName: request.Name,
            email: request.Email,
            password: passwordHasher.Hash(request.Password),
            birthDate: request.BirthDate,
            gender: request.Gender,
            weight: request.Weight,
            height: request.Height,
            caloriePlan: request.CaloriePlan,
            carbonPercent: request.CarbonPercent,
            proteinPercent: request.ProteinPercent,
            fatPercent: request.FatPercent,
            sportLevel: request.SportLevel,
            registrationDate: clock.UtcNow);

        await userRep.AddAsync(user, ct);
        userProvider.SetUser(user.Id);
        return user;
    }

    public async Task<User> LoginAsync(string email, string password, CancellationToken ct = default)
    {
        var user = await userRep.GetByEmailAsync(email, ct)
            ?? throw new UserNotFoundException(email); 

        if (!passwordHasher.Verify(password, user.Password))
            throw new InvalidPasswordException();

        userProvider.SetUser(user.Id);
        return user;
    }

    public async Task<bool> IsUserExistsAsync(string email, CancellationToken ct = default) =>
        await userRep.ExistsByEmailAsync(email, ct);

    public void Logout()
    {
        userProvider.Clear();
        userService.Clear();
    }

    public async Task ChangePasswordAsync(string newPassword, CancellationToken ct = default)
    {
        var passwordErrors = PasswordPolicy.Validate(newPassword);

        if (passwordErrors.Any())
            throw new WeakPasswordException(passwordErrors); 

        var user = await userRep.GetByIdAsync(userProvider.RequireUserId(), ct)
            ?? throw new UserNotFoundException(); 

        user.ChangePassword(passwordHasher.Hash(newPassword));
        await userRep.UpdateAsync(user, ct);
    }

    public async Task UpdateProfileAsync(User user, CancellationToken ct = default)
    {
        await userRep.UpdateAsync(user, ct);
        userService.Clear();
    }

    public async Task DeleteProfileAsync(CancellationToken ct = default)
    {
        await   userRep.DeleteAsync(userProvider.RequireUserId(), ct);
        Logout();
    }
}

using StepUp.Application.Exceptions;
using StepUp.Application.Interfaces.Repositories;
using StepUp.Application.Interfaces.Services;
using StepUp.Domain.Enums;

namespace StepUp.Application.Services;

public class DayResultService(IDayResultRepository dayResults,
                        ICurrentUserProvider userProvider,
                        ICurrentUserService userService,
                        IUserRepository users,
                        IClock clock)
{

    private async Task<DayResult> GetOrCreateAsync(DateTime date, CancellationToken ct = default)
    {
        var user = await userService.GetOrLoadAsync(ct) ??
            throw new UserNotFoundException();

        var todayResult = await dayResults.GetAsync(user.Id, date, ct);
        if (todayResult != null) return todayResult;

        todayResult = new DayResult(
            userId: user.Id,
            caloriePlan: user.CaloriePlan,
            date: date,
            weight: user.Weight);

        await dayResults.AddAsync(todayResult, ct);
        return todayResult;
    }

    public async Task<DayResult> GetOrCreateTodayAsync(CancellationToken ct = default)
    {
        return await GetOrCreateAsync(clock.Today, ct);
    }
    public async Task<DayResult> GetOrCreateDayAsync(DateTime date, CancellationToken ct = default)
    {
        return await GetOrCreateAsync(date, ct);
    }
    public async Task<DayResult?> GetDayAsync(DateTime date, CancellationToken ct = default)
    {
        return await dayResults.GetAsync(userProvider.RequireUserId(), date, ct);
    }

    public async Task<DayResult> AddFoodAsync(DateTime date, MealType mealType, FoodProduct product, Serving serving,
                                   double amount, CancellationToken ct = default)
    {
        var day = await GetDayAsync(date, ct)
            ?? throw new DayResultNotFoundException();

        ArgumentOutOfRangeException.ThrowIfNegativeOrZero(amount);
        day.AddToMeal(mealType, product, amount, serving);
        await dayResults.UpdateAsync(day, ct);

        return day;
    }

    public async Task<DayResult> RemoveFoodAsync(DateTime date, MealType mealType,
        Guid foodItemId, CancellationToken ct = default)
    {
        var day = await GetDayAsync(date, ct)
            ?? throw new DayResultNotFoundException(); 

        day.RemoveFromMeal(mealType, foodItemId);
        await dayResults.UpdateAsync(day, ct);

        return day;
    }

    public async Task<DayResult> ClearMealAsync(DateTime date, MealType mealType, CancellationToken ct = default)
    {
        var day = await GetDayAsync(date, ct)
            ?? throw new DayResultNotFoundException();

        day.ClearMeal(mealType);
        await dayResults.UpdateAsync(day, ct);

        return day;
    }

    public async Task<DayResult> ChangeFoodAmountAsync(DateTime date, MealType mealType, Guid itemId,
        double newAmount, Serving? newServing = null, CancellationToken ct = default)
    {
        var day = await GetDayAsync(date, ct)
            ?? throw new DayResultNotFoundException();

        ArgumentOutOfRangeException.ThrowIfNegativeOrZero(newAmount);

        var item = day.GetItemFromMeal(mealType, itemId)
            ?? throw new FoodItemNotFoundException();

        if (newServing != null) item.ChangeServing(newServing);
        item.ChangeAmount(newAmount);

        await dayResults.UpdateAsync(day, ct);

        return day;
    }

    public async Task<DayResult> AddSportAsync(DateTime date, string name, int kcalBurnt, CancellationToken ct = default)
    {
        var day = await GetDayAsync(date, ct)
            ?? throw new DayResultNotFoundException(); 

        day.AddSport(name, kcalBurnt);
        await dayResults.UpdateAsync(day, ct);

        return day;
    }

    public async Task<DayResult> RemoveSportAsync(DateTime date, Guid sportId, CancellationToken ct = default)
    {
        var day = await GetDayAsync(date, ct)
            ?? throw new DayResultNotFoundException(); 

        day.RemoveSport(sportId);
        await dayResults.UpdateAsync(day, ct);

        return day;
    }

    public async Task<DayResult> EditSportAsync(DateTime date, Guid sportId,
        string newName, int kcalBurnt, CancellationToken ct = default)
    {
        var day = await GetDayAsync(date, ct)
           ?? throw new DayResultNotFoundException();

        if (day.GetSport(sportId) is null)
            throw new SportNotFoundException();

        day.EditSport(sportId, newName, kcalBurnt);
        await dayResults.UpdateAsync(day, ct);

        return day;
    }

    public async Task<DayResult> UpdateWeightAsync(double newWeight, CancellationToken ct = default)
    {
        var user = await userService.GetOrLoadAsync(ct) ??
            throw new UserNotFoundException();

        user.ChangeWeight(newWeight);

        userService.Clear();

        todayResult.ChangeWeight(newWeight);
        await dayResults.UpdateAsync(todayResult, ct);

        return todayResult;
    }

    public async Task<DayResult> UpdateCaloriePlanAsync(int newCaloriePlan, CancellationToken ct = default)

        var user = await userService.GetOrLoadAsync(ct) ??
            throw new UserNotFoundException();

        var todayResult = await GetOrCreateTodayAsync(ct);

        user.ChangeCaloriePlan(newCaloriePlan);
        await users.UpdateAsync(user, ct);
        userService.Clear();

        todayResult.ChangeCaloriePlan(newCaloriePlan);
        await dayResults.UpdateAsync(todayResult, ct);

        return todayResult ;
    }

using System.Text.RegularExpressions;

namespace StepUp.Application.Services;

public static class PasswordPolicy
{
    public const int MinLength = 8;

    public static List<string> Validate(string password)
    {
        var errors = new List<string>();

        if (string.IsNullOrWhiteSpace(password) || password.Length < MinLength)
            errors.Add($"Пароль должен содержать минимум {MinLength} символов.");

        if (!Regex.IsMatch(password, @"[a-z]"))
            errors.Add("Пароль должен содержать хотя бы одну строчную букву.");

        if (!Regex.IsMatch(password, @"[A-Z]"))
            errors.Add("Пароль должен содержать хотя бы одну заглавную букву.");

        if (!Regex.IsMatch(password, @"\d"))
            errors.Add("Пароль должен содержать хотя бы одну цифру.");

        if (!Regex.IsMatch(password, @"[^a-zA-Z0-9]"))
            errors.Add("Пароль должен содержать хотя бы один спецсимвол (!@#$%^&* и т.д.).");

        return errors;
    }

    public static bool IsValid(string password) => Validate(password).Count == 0;
}

    {
        _db.DayResults.Add(day);
        await _db.SaveChangesAsync(ct);
    }

    public async Task UpdateAsync(DayResult day, CancellationToken ct = default)
    {
        if (_db.Entry(day).State == EntityState.Detached)
            _db.DayResults.Update(day);

        await _db.SaveChangesAsync(ct);
    }

    public Task DeleteAsync(Guid dayId, CancellationToken ct = default) =>
        _db.DayResults
            .Where(d => d.Id == dayId)
            .ExecuteDeleteAsync(ct);

using StepUp.Domain.Entities;

namespace StepUp.Domain.Calculators;

public static class MacroCalculator
{
    public static double CalculateProteinCarbonGrams(int caloriePlan, int proteinPercent)
    {
        if (caloriePlan <= 0 || proteinPercent <= 0) return 0;

        double caloriesFromProtein = caloriePlan * (proteinPercent / 100.0);
        return Math.Round(caloriesFromProtein / 4.0, 1);
    }

    public static double CalculateFatGrams(int caloriePlan, int fatPercent)
    {
        if (caloriePlan <= 0 || fatPercent <= 0) return 0;

        double caloriesFromFat = caloriePlan * (fatPercent / 100.0);
        return Math.Round(caloriesFromFat / 9.0, 1);
    }

    public static (double Protein, double Fat, double Carbon) CalculateMacroPercentages(
        double proteins, double fats, double carbohydrates)
    {
        double proteinKcal = proteins * 4.0;
        double fatKcal = fats * 9.0;
        double carbonKcal = carbohydrates * 4.0;

        double totalKcal = proteinKcal + fatKcal + carbonKcal;

        if (totalKcal <= 0)
            return (0, 0, 0);

        double proteinPct = Math.Round(proteinKcal / totalKcal * 100, 1);
        double fatPct = Math.Round(fatKcal / totalKcal * 100, 1);

        double carbonPct = Math.Round(100.0 - proteinPct - fatPct, 1);

        return (proteinPct, fatPct, carbonPct);
    }

    public static (double Protein, double Fat, double Carbon) CalculateMacroPercentages(Nutrients nutrients)
        => CalculateMacroPercentages(nutrients.Proteins, nutrients.Fat, nutrients.Carbohydrates);

}

    {
        if (amount < 0)

            < 24.9 => BmiCategory.Normal,
            < 29.9 => BmiCategory.Overweight,
            < 34.9 => BmiCategory.Obese1,
            < 39.9 => BmiCategory.Obese2,
            _ => BmiCategory.Obese3
        };

        return new Bmi(value, category);
    }

using StepUp.Application.Interfaces;
using StepUp.Domain.Entities;
using System.Globalization;
using System.Text.Json;
using System.Text.RegularExpressions;

namespace StepUp.Infrastructure.OpenFoodFacts;

public class OpenFoodFactsClient(HttpClient httpClient) : IFoodProductClient
{
    private readonly HttpClient _httpClient = httpClient;

    public async Task<FoodProduct?> GetByBarcodeAsync(string barcode,
        CancellationToken ct = default)
    {
        if (string.IsNullOrWhiteSpace(barcode))
            return null;
        try
        {
            var response = await _httpClient.GetAsync($"/api/v0/product/{barcode}.json", ct);
            response.EnsureSuccessStatusCode();

            var json = await response.Content.ReadAsStringAsync(ct);
            var doc = JsonDocument.Parse(json);
            var root = doc.RootElement;

            return MapProduct(root.GetProperty("product"), barcode);
        }
        catch (HttpRequestException)
        {
            return null;
        }
        catch (JsonException)
        {
            return null;
        }
        catch (OperationCanceledException)
        {
            throw;
        }
    }

    public async Task<List<FoodProduct>> SearchByNameAsync(
        string query, CancellationToken ct = default)
    {
        if (string.IsNullOrWhiteSpace(query))
            return [];
        if (Regex.IsMatch(query.Trim(), @"^\d+$"))
        {
            var product = await GetByBarcodeAsync(query, ct);
            if (product == null) return [];
            return [product];
        }

        try
        {
            var url = "cgi/search.pl" +
                      $"?search_terms={Uri.EscapeDataString(query)}" +
                      "&search_simple=1&action=process&json=1&page_size=20";

            var response = await _httpClient.GetAsync(url, ct);
            response.EnsureSuccessStatusCode();

            var json = await response.Content.ReadAsStringAsync(ct);
            using var doc = JsonDocument.Parse(json);
            var root = doc.RootElement;

            if (!root.TryGetProperty("products", out var products) ||
                products.GetArrayLength() == 0)
                return [];

            var result = new List<FoodProduct>();
            foreach (var p in products.EnumerateArray())
            {
                var barcode = p.TryGetProperty("code", out var codeEl)
                    ? codeEl.GetString()
                    : null;

                var product = MapProduct(p, barcode ?? string.Empty);
                if (product is null)
                    continue;

                result.Add(product);
                if (result.Count >= 10)
                    break;
            }
            return result;
        }
        catch (HttpRequestException) { return []; }
        catch (JsonException) { return []; }
        catch (OperationCanceledException) { throw; }
    }

    private FoodProduct? MapProduct(JsonElement product, string barcode)
    {
        if (!product.TryGetProperty("nutriments", out var n))
            return null;
        var calories = ParseDouble(n, "energy-kcal_100g");
        if (calories <= 0)
            return null;
        return new FoodProduct
        (
            id: barcode ?? Guid.NewGuid().ToString(),
            name: product.TryGetProperty("product_name", out var name) ? name.GetString() : null,
            barcode: barcode,
            nutrientsPer100g: new Nutrients((int)Math.Round(calories, 0),
                                            ParseDouble(n, "proteins_100g"),
                                            ParseDouble(n, "fat_100g"),
                                            ParseDouble(n, "carbohydrates_100g")),
            mainServing: MapServing(product),
            brands: product.TryGetProperty("brands", out var brandsElement) ? brandsElement.GetString() : null,
            nutriscoreGrade: product.TryGetProperty("nutriscore_grade", out var nsg) ? nsg.GetString() : null,
            novaGroup: ParseIntToString(product, "nova_group"),
            imageUrl: product.TryGetProperty("image_url", out var image) ?
                image.GetString() : $"{_httpClient.BaseAddress}/images/icons/dist/packaging.svg",
            sugars100g: ParseDouble(n, "sugars_100g")
        );
    }
    private static double ParseDouble(JsonElement element, string property)
    {
        if (element.ValueKind == JsonValueKind.Undefined)
            return 0;

        if (!element.TryGetProperty(property, out var value))
            return 0;

        return value.ValueKind switch
        {
            JsonValueKind.Number => value.GetDouble(),
            JsonValueKind.String when double.TryParse(
                value.GetString(),
                NumberStyles.Any,
                CultureInfo.InvariantCulture,
                out var d) => d,
            _ => 0
        };
    }
    private static string? ParseIntToString(JsonElement element, string property)
    {
        if (element.ValueKind == JsonValueKind.Undefined)
            return null;

        if (!element.TryGetProperty(property, out var value))
            return null;

        return value.ValueKind switch
        {
            JsonValueKind.Number => value.GetInt32().ToString(),
            JsonValueKind.String when int.TryParse(
                value.GetString(),
                out var d) => d.ToString(),
            _ => null
        };
    }
    private static Serving? MapServing(JsonElement product)
    {
        var solidUnits = new Dictionary<string, double>()
        {
            ["mg"] = 0.001,
            ["g"] = 1,
            ["kg"] = 1000,
            ["oz"] = 28.3495,
            ["lb"] = 453.592,
            ["мг"] = 0.001,
            ["г"] = 1,
            ["кг"] = 1000,
        };
        var liquidUnits = new Dictionary<string, double>()
        {
            ["мл"] = 1,
            ["л"] = 1000,
            ["ml"] = 1,
            ["cl"] = 10,
            ["dl"] = 100,
            ["l"] = 1000,
            ["pt"] = 473.176,
            ["qt"] = 946.353,
            ["fl oz"] = 29.5735,
            ["gal"] = 3785.41,
        };
        var q = ParseDouble(product, "serving_quantity");
        if (product.TryGetProperty("serving_quantity_unit", out var qu) && q >= 0)
        {
            var qus = qu.GetString();
            if (qus != null)
            {
                qus = qus.ToLower();
                if (liquidUnits.TryGetValue(qus, out double amount))
                {
                    return new Serving("порция", amount * q, true);
                }
                else if (solidUnits.TryGetValue(qus, out amount))
                {
                    return new Serving("порция", amount * q);
                }
            }
        }
        q = ParseDouble(product, "product_quantity");
        if (product.TryGetProperty("product_quantity_unit", out qu) && q >= 0)
        {
            var qus = qu.GetString();
            if (qus != null)
            {
                qus = qus.ToLower();
                if (liquidUnits.TryGetValue(qus, out double amount))
                {
                    return new Serving("упаковка", amount * q, true);
                }
                else if (solidUnits.TryGetValue(qus, out amount))
                {
                    return new Serving("упаковка", amount * q);
                }
            }
        }
        return null;
    }

public class BCryptPasswordHasher : IPasswordHasher
{
    private const int WorkFactor = 12;

    public string Hash(string password)
    {
        if (string.IsNullOrWhiteSpace(password))
            throw new ArgumentException("Пароль не может быть пустым.", nameof(password));

        return BCrypt.Net.BCrypt.HashPassword(password, WorkFactor);
    }

    public bool Verify(string password, string hash)
    {
        if (string.IsNullOrWhiteSpace(password) || string.IsNullOrWhiteSpace(hash))
            return false;

        try
        {
            return BCrypt.Net.BCrypt.Verify(password, hash);
        }
        catch (BCrypt.Net.SaltParseException)
        {
            return false;
        }
    }
}

public class MacroPieChartDrawable : IDrawable
{
    public float ProteinPercent { get; set; }
    public float FatPercent { get; set; }
    public float CarbonPercent { get; set; }

    private float _centerX = 70;
    private float _centerY = 70;
    private float _radius;
    private ICanvas _canvas = null!;

    public Color ProteinColor { get; set; } = Color.FromArgb("#9fd5bd");
    public Color FatColor { get; set; } = Color.FromArgb("#fceda8");
    public Color CarbonColor { get; set; } = Color.FromArgb("#e79bc3");
    public float StrokeWidth { get; set; } = 20f;

    public void Draw(ICanvas canvas, RectF dirtyRect)
    {
        _canvas = canvas;
        _centerX = dirtyRect.Width / 2f;
        _centerY = dirtyRect.Height / 2f;
        _radius = Math.Min(_centerX, _centerY) - StrokeWidth / 2f;
        if (_radius <= 0) return;

        canvas.StrokeSize = StrokeWidth;
        canvas.StrokeLineCap = LineCap.Butt;

        var startAngle = -90f;

        DrawSlice(ref startAngle, ProteinPercent, ProteinColor);
        DrawSlice(ref startAngle, FatPercent, FatColor);
        DrawSlice(ref startAngle, CarbonPercent, CarbonColor);
    }

    private void DrawSlice(ref float startAngle, float percent, Color color)
    {
        if (percent <= 0) return;
        _canvas.StrokeColor = color;
        _canvas.StrokeSize = 15;
        if (percent == 100)
        {
            _canvas.DrawCircle(_centerX, _centerY, _radius);
            return;
        }

        var sweep = percent * 3.6f; 
        _canvas.DrawArc(
            _centerX - _radius, _centerY - _radius,
            _radius * 2, _radius * 2,
            startAngle, startAngle + sweep,
            clockwise: true,
            closed: false);

        startAngle += sweep;
    }
}

using System.Globalization;

namespace StepUp.Maui.Converters;

public class MacroRatioToColorConverter : IValueConverter
{
    public object? Convert(object? value, Type targetType, object? parameter, CultureInfo culture)
    {
        if (value is not double ratio)
            return Colors.Gray;

        if (ratio < 0) ratio = 0;

        Color green = Colors.Green;
        Color yellow = Colors.Yellow;
        Color orange = Colors.Orange;
        Color red = Colors.Red;

        return ratio switch
        {
            <= 0.5 => Interpolate(green, yellow, ratio / 0.5),            
            <= 0.85 => Interpolate(yellow, orange, (ratio - 0.5) / 0.35), 
            <= 1.0 => Interpolate(orange, red, (ratio - 0.85) / 0.15),    
            _ => Colors.DarkRed                                       
        };
    }

    private static Color Interpolate(Color start, Color end, double progress)
    {
        return new Color(
            (float)(start.Red + (end.Red - start.Red) * progress),
            (float)(start.Green + (end.Green - start.Green) * progress),
            (float)(start.Blue + (end.Blue - start.Blue) * progress)
        );
    }

    public object? ConvertBack(object? value, Type targetType, object? parameter, CultureInfo culture)
        => throw new NotImplementedException();
}

        IMTIndicator.Margin = new Thickness(relativeOffset, 0, 0, 0);
        IMTIndicator.IsVisible = true;

    {
        double targetEaten = _vm.KcalEaten;
        double targetBurnt = _vm.KcalBurnt;

        _vm.KcalEaten = 0;
        _vm.KcalBurnt = 0;

        new Animation(
            callback: v =>
            {
                _vm.KcalEaten = (int)Math.Round(v * targetEaten);
                _vm.KcalBurnt = (int)Math.Round(v * targetBurnt);
            },
            start: 0,
            end: 1,
            easing: Easing.CubicOut
        ).Commit(this, "AnimAll", 16, 1500);

        AnimateMacros();
        AnimateKcalCircle();
    }

    private void AnimateMacros()
    {
        double targetPr = _vm.PrProgress;
        double targetFat = _vm.FatProgress;
        double targetCb = _vm.CbProgress;

        _vm.PrProgress = 0;
        _vm.FatProgress = 0;
        _vm.CbProgress = 0;

        var animation = new Animation(
            callback: v =>
            {
                _vm.PrProgress = v * targetPr;
                _vm.FatProgress = v * targetFat;
                _vm.CbProgress = v * targetCb;
            },
            start: 0,
            end: 1,
            easing: Easing.CubicOut 
        );

        animation.Commit(this, "MacrosAnim", 16, 1000);
    }

    private void AnimateKcalCircle()
    {
        double targetProgress = _vm.KcalProgress;
        int targetKcalRes = _vm.KcalRes;

        progressBar.IsVisible = targetProgress != 0;
        Arc.Point = new Point(67, 0);
        Arc.IsLargeArc = false;
        progressBar.Stroke = new SolidColorBrush(Colors.Green);
        KcalResLabel.Text = "0";
        KcalLeft.TextColor = Colors.Green;


        var animation = new Animation(
            callback: v =>
            {
                double currentProgress = Math.Min(v * targetProgress, 0.9999);

                double angle = (Math.PI * 2 * currentProgress) - (Math.PI / 2);
                double x = 67 + 67 * Math.Cos(angle);
                double y = 67 + 67 * Math.Sin(angle);

                Arc.Point = new Point(x, y);
                Arc.IsLargeArc = currentProgress > 0.5;
                Arc.SweepDirection = SweepDirection.Clockwise;

                var currentColor = ColorInterpolator.GetColor(currentProgress);
                progressBar.Stroke = new SolidColorBrush(currentColor);
                KcalResLabel.TextColor = currentColor;
                KcalLeft.TextColor = currentColor;

                KcalResLabel.Text = ((int)(v * targetKcalRes)).ToString();
            },
            start: 0,
            end: 1,
            easing: Easing.CubicOut 
        );

        animation.Commit(this, "KcalCircleAnim", 16, 1500);
    }
}

using Android.Content;
using Android.Preferences;

namespace StepUp.Maui.Platforms.Android.Services;
public class DailyStepCounter
{
    private const string StepsKey = "daily_steps";
    private const string DateKey = "last_step_date";

    private readonly ISharedPreferences _prefs;

    public DailyStepCounter(Context context)
    {
        _prefs = PreferenceManager.GetDefaultSharedPreferences(context);
    }

    public void SaveSteps(int totalSteps, int todaySteps)
    {
        var editor = _prefs.Edit();
        editor.PutInt(StepsKey, totalSteps);
        editor.PutInt(DateKey, DateTime.Now.DayOfYear);
        editor.PutInt("today_steps", todaySteps);
        editor.Apply();
    }

    public (int todaySteps, bool isNewDay) GetDailySteps(int currentSensorSteps)
    {
        var lastDate = _prefs.GetInt(DateKey, -1);
        var lastSteps = _prefs.GetInt(StepsKey, 0);
        var savedTodaySteps = _prefs.GetInt("today_steps", 0);

        var isNewDay = DateTime.Now.DayOfYear != lastDate;

        if (isNewDay)
        {
            return (0, true); 
        }

        var stepsSinceLastSave = currentSensorSteps - lastSteps;
        return (savedTodaySteps + stepsSinceLastSave, false);
    }
}

namespace StepUp.Maui.Utils;

public static class ColorInterpolator
{
    private static readonly Color[] UsedColors =
    {
        Colors.Green,
        Colors.Yellow,
        Colors.Orange,
        Colors.Red
    };

    public static Color GetColor(double progress)
    {
        progress = Math.Clamp(progress, 0, 1);

        int index = (int)(progress * (UsedColors.Length - 1));
        double localProgress = (progress * (UsedColors.Length - 1)) - index;

        if (index >= UsedColors.Length - 1) return UsedColors[^1];

        Color start = UsedColors[index];
        Color end = UsedColors[index + 1];

        return new Color(
            (float)(start.Red + (end.Red - start.Red) * localProgress),
            (float)(start.Green + (end.Green - start.Green) * localProgress),
            (float)(start.Blue + (end.Blue - start.Blue) * localProgress)
        );
    }
}

    private void ValidateData()
    {
        ErrorMessageVisible = false;

        bool validW = double.TryParse(WeightInput, out var w) && w >= 30 && w <= 300;
        bool validH = double.TryParse(HeightInput, out var h) && h >= 100 && h <= 250;

        bool validP = int.TryParse(ProteinInput, out var p) && p > 0;
        bool validF = int.TryParse(FatInput, out var f) && f > 0;

            IsFormValid = false;
            IsPfcValid = false;
        }
        else
        {
            IsFormValid = true;
            IsPfcValid = true;
        }
        ContinueCommand.NotifyCanExecuteChanged();
    }

    [RelayCommand(CanExecute = nameof(CanContinue), AllowConcurrentExecutions = false)]
    private async Task ContinueAsync()
    {
        if (!IsStepTwo)
        {
            CaloriePlan = CalculateCalories();
            IsStepTwo = true;
            ValidatePfc();
        }
        else
        {
            if (_request is null) return;

            _request.Weight = double.Parse(WeightInput);
            _request.Height = int.Parse(HeightInput);
            _request.SportLevel = SelectedSportLevel.Value;
            _request.CaloriePlan = CaloriePlan;
            _request.ProteinPercent = int.Parse(ProteinInput);
            _request.FatPercent = int.Parse(FatInput);
            _request.CarbonPercent = int.Parse(CarbonInput);

            try
            {
                await authService.RegisterAsync(_request);
                await navigationService.GoToPageAsync(AppRoutes.Main);
            }
            catch
            {
                ErrorMessageVisible = true;
            }
        }
    }

    private bool CanContinue()
    {
        if (!IsFormValid) return false;

        if (IsStepTwo)
        {
            return int.TryParse(ProteinInput, out var p) && p > 0 &&
                   int.TryParse(FatInput, out var f) && f > 0 &&
                   int.TryParse(CarbonInput, out var c) && c > 0 &&
                   (p + f + c == 100);
        }
        return true;
    }

    private async Task BackAsync()
    {
        ErrorMessageVisible = false;
        if (!IsStepTwo)
        {
            await navigationService.GoBackAsync();
        }
        else
        {
            IsStepTwo = false;
            ValidateData();
        }
    }

    [RelayCommand]
    private async Task EditCaloriesAsync()
    {
        ErrorMessageVisible = false;
        var defaultKcal = CaloriePlan.ToString();

        var input = await dialogService.ShowPromptAsync("Введите желаемый калораж", "Значение не должно быть меньше 1300 ккал", defaultKcal);

        if (string.IsNullOrEmpty(input)) return;

        if (int.TryParse(input, out int newKcal) && newKcal >= 1300)
        {
            CaloriePlan = newKcal;
        }
        else
        {
            await dialogService.ShowAlertAsync("Ошибка", "Калораж введён неверно! Попробуйте снова.");
        }
    }

    private int CalculateCalories()
    {
        if (_request is null) return 2000;

        var today = clock.Today;
        int age = today.Year - _request.BirthDate.Year;
        if (_request.BirthDate.Date > today.AddYears(-age)) age--;

        double ko = SelectedSportLevel.Value switch
        {
            SportLevel.No => 1.2,
            SportLevel.Low => 1.375,
            SportLevel.Mid => 1.55,
            SportLevel.High => 1.7,
            SportLevel.Extreme => 1.9,
            _ => 1.2
        };

        double res;
        if (_request.Gender == Gender.Male)
            res = 66.5 + (13.75 * double.Parse(WeightInput)) + (5.003 * double.Parse(HeightInput)) - (6.775 * age);
        else
            res = 655.1 + (9.563 * double.Parse(WeightInput)) + (1.85 * double.Parse(HeightInput)) - (4.676 * age);

        return (int)(res * ko);
    }

using System.ComponentModel;
using System.Runtime.CompilerServices;

namespace StepUp.Presentation.ViewModels;

public class StepViewModel : INotifyPropertyChanged
{
    private string _stepsText = "0 шагов";

    public string StepsText
    {
        get => _stepsText;
        private set
        {
            if (_stepsText == value) return;
            _stepsText = value;
            OnPropertyChanged();
        }
    }

    public string Calories
    {
        get => (int.Parse(_stepsText.Split(' ')[0]) * 0.05).ToString() + " ккал";
    }

    public StepViewModel()
    {
        StepCounterService.StepsUpdated += steps =>
        {
            StepsText = FormatSteps(steps);
        };
    }


    private string FormatSteps(int count)
    {
        if (count == 0) return "0 шагов";

        string stepWord = "шагов";
        if (count % 10 == 1 && count % 100 != 11)
            stepWord = "шаг";
        else if (count % 10 >= 2 && count % 10 <= 4 &&
                !(count % 100 >= 12 && count % 100 <= 14))
            stepWord = "шага";

        return $"{count} {stepWord}";
    }
    public void RefreshAll()
    {
        OnPropertyChanged(nameof(StepsText));
        OnPropertyChanged(nameof(Calories));
    }
    public event PropertyChangedEventHandler PropertyChanged;
    protected void OnPropertyChanged([CallerMemberName] string name = null) =>
        PropertyChanged?.Invoke(this, new PropertyChangedEventArgs(name));
}
